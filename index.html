// VPI Dashboard — Google Apps Script (v40) // Sheet names (exact — verified from live Google Sheet tabs): // MR Weekly, MR Daily, Downtime, Prod Cost, MR Monthly, PC daily, MCOS daily // Sites: AC, PFMIS, HOREB, BUKID, ARGAO, CCPC, SOUTH (+ National aggregate) // Deploy as: Web App → Execute as Me → Anyone // ─── SHEET NAME CONSTANTS ──────────────────────────────────────────────────── const SHEET = { WEEKLY: "MR Weekly", DAILY: "MR Daily", DOWNTIME: "Downtime", PROD_COST:"Prod Cost", MONTHLY: "MR Monthly", PC_DAILY: "PC daily", MCOS: "MCOS daily" }; const SITES = ["AC", "PFMIS", "HOREB", "BUKID", "ARGAO", "CCPC", "SOUTH"]; // ─── ENTRY POINT ───────────────────────────────────────────────────────────── function doGet(e) { const params = e && e.parameter ? e.parameter : {}; const tab = params.tab || "weekly"; const site = params.site || "National"; const week = params.week || ""; const month = params.month || ""; const callback = params.callback || ""; // JSONP support let result; try { switch (tab) { case "weekly": result = getWeeklyPulse(site, week); break; case "monthly": result = getMonthly(site, month); break; case "cost": result = getCost(site, week); break; case "downtime": result = getDowntime(site, week); break; case "production": result = getProduction(site, week); break; case "oee": result = getOEE(site, week); break; case "cost_analytics": result = getCostAnalytics(site, month); break; case "quality_energy": result = getQualityEnergy(site, week); break; case "meta": result = getMeta(); break; default: result = getWeeklyPulse(site, week); } } catch (err) { result = { error: err.message, tab, site }; } const json = JSON.stringify(result); // JSONP callback (for browsers that block direct fetch) if (callback) { return ContentService .createTextOutput(callback + "(" + json + ");") .setMimeType(ContentService.MimeType.JAVASCRIPT); } // Plain JSON with CORS headers return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } // Required for CORS preflight function doOptions(e) { return ContentService .createTextOutput("") .setMimeType(ContentService.MimeType.TEXT); } // ─── HELPERS ───────────────────────────────────────────────────────────────── function getSheet(name) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(name); if (!sheet) throw new Error("Sheet not found: " + name); return sheet; } function sheetToObjects(sheet, headerRow) { headerRow = headerRow || 1; const data = sheet.getDataRange().getValues(); const headers = data[headerRow - 1].map(h => String(h).trim()); const rows = []; for (let i = headerRow; i < data.length; i++) { if (data[i].every(c => c === "" || c === null)) continue; const obj = {}; headers.forEach((h, j) => { obj[h] = data[i][j]; }); rows.push(obj); } return rows; } function filterBySite(rows, site, siteCol) { siteCol = siteCol || "Site"; if (!site || site === "National") return rows; return rows.filter(r => String(r[siteCol]).trim().toUpperCase() === site.toUpperCase()); } function filterByWeek(rows, week, weekCol) { weekCol = weekCol || "Week"; if (!week) return rows; return rows.filter(r => String(r[weekCol]).trim() === String(week).trim()); } function filterByMonth(rows, month, monthCol) { monthCol = monthCol || "Month"; if (!month) return rows; return rows.filter(r => String(r[monthCol]).trim() === String(month).trim()); } function sumField(rows, field) { return rows.reduce((s, r) => s + (parseFloat(r[field]) || 0), 0); } function uniqueWeeks(rows, weekCol) { weekCol = weekCol || "Week"; return [...new Set(rows.map(r => r[weekCol]).filter(Boolean))].sort(); } function uniqueMonths(rows, monthCol) { monthCol = monthCol || "Month"; return [...new Set(rows.map(r => r[monthCol]).filter(Boolean))].sort(); } // ─── TAB HANDLERS ───────────────────────────────────────────────────────────── // 1. Weekly Pulse (MR Weekly) — returns ALL rows so dashboard can filter by week client-side function getWeeklyPulse(site, week) { const sheet = getSheet(SHEET.WEEKLY); const allRows = sheetToObjects(sheet); const weeks = uniqueWeeks(allRows); if (!week) week = weeks[weeks.length - 1] || ""; // Return ALL rows (all weeks, all sites) so the dashboard can filter freely // This enables week selector to work for weeks 1-21 return { tab: "weekly", site, week, weeks, rows: allRows // ALL rows — dashboard filters client-side }; } // 2. Monthly (MRMonthly) function getMonthly(site, month) { const sheet = getSheet(SHEET.MONTHLY); let rows = sheetToObjects(sheet); const months = uniqueMonths(rows); if (!month) month = months[months.length - 1] || ""; rows = filterByMonth(filterBySite(rows, site), month); return { tab: "monthly", site, month, months, rows, totals: { production: sumField(rows, "Production"), cost: sumField(rows, "Cost"), downtime: sumField(rows, "Downtime") } }; } // 3. Cost (Prod Cost) function getCost(site, week) { const sheet = getSheet(SHEET.PROD_COST); let rows = sheetToObjects(sheet); const weeks = uniqueWeeks(rows); if (!week) week = weeks[weeks.length - 1] || ""; rows = filterByWeek(filterBySite(rows, site), week); return { tab: "cost", site, week, weeks, rows, totals: { total_cost: sumField(rows, "Total Cost"), feed_cost: sumField(rows, "Feed Cost"), labor_cost: sumField(rows, "Labor Cost"), utility_cost: sumField(rows, "Utility Cost"), other_cost: sumField(rows, "Other Cost") } }; } // 4. Downtime (Downtime) function getDowntime(site, week) { const sheet = getSheet(SHEET.DOWNTIME); let rows = sheetToObjects(sheet); const weeks = uniqueWeeks(rows); if (!week) week = weeks[weeks.length - 1] || ""; rows = filterByWeek(filterBySite(rows, site), week); // Group by reason const byReason = {}; rows.forEach(r => { const reason = r["Reason"] || r["Downtime Reason"] || "Unknown"; byReason[reason] = (byReason[reason] || 0) + (parseFloat(r["Hours"] || r["Downtime Hours"]) || 0); }); return { tab: "downtime", site, week, weeks, rows, byReason, totals: { total_hours: sumField(rows, "Hours") || sumField(rows, "Downtime Hours") } }; } // 5. Production (MR Daily + MR Weekly) function getProduction(site, week) { const weeklySheet = getSheet(SHEET.WEEKLY); const dailySheet = getSheet(SHEET.DAILY); let weeklyRows = sheetToObjects(weeklySheet); let dailyRows = sheetToObjects(dailySheet); const weeks = uniqueWeeks(weeklyRows); if (!week) week = weeks[weeks.length - 1] || ""; weeklyRows = filterByWeek(filterBySite(weeklyRows, site), week); dailyRows = filterByWeek(filterBySite(dailyRows, site), week); return { tab: "production", site, week, weeks, weekly: weeklyRows, daily: dailyRows, totals: { total_production: sumField(weeklyRows, "Production"), target_production: sumField(weeklyRows, "Target"), actual_vs_target: ( sumField(weeklyRows, "Target") > 0 ? ((sumField(weeklyRows, "Production") / sumField(weeklyRows, "Target")) * 100).toFixed(1) : 0 ) } }; } // 6. OEE (MR Weekly + MR Daily) function getOEE(site, week) { const sheet = getSheet(SHEET.WEEKLY); let rows = sheetToObjects(sheet); const weeks = uniqueWeeks(rows); if (!week) week = weeks[weeks.length - 1] || ""; rows = filterByWeek(filterBySite(rows, site), week); const avgOEE = rows.length ? (sumField(rows, "OEE") / rows.length).toFixed(2) : 0; // Per-site breakdown for National view const bySite = {}; if (!site || site === "National") { SITES.forEach(s => { const sRows = filterBySite(rows, s); bySite[s] = sRows.length ? (sumField(sRows, "OEE") / sRows.length).toFixed(2) : 0; }); } return { tab: "oee", site, week, weeks, rows, bySite, totals: { avg_oee: avgOEE, availability: rows.length ? (sumField(rows, "Availability") / rows.length).toFixed(2) : 0, performance: rows.length ? (sumField(rows, "Performance") / rows.length).toFixed(2) : 0, quality_rate: rows.length ? (sumField(rows, "Quality Rate") / rows.length).toFixed(2) : 0 } }; } // 7. Cost Analytics (PC Daily + Prod Cost) function getCostAnalytics(site, month) { const pcDailySheet = getSheet(SHEET.PC_DAILY); const prodCostSheet = getSheet(SHEET.PROD_COST); let pcRows = sheetToObjects(pcDailySheet); let costRows = sheetToObjects(prodCostSheet); const months = uniqueMonths(pcRows); if (!month) month = months[months.length - 1] || ""; pcRows = filterByMonth(filterBySite(pcRows, site), month); costRows = filterBySite(costRows, site); return { tab: "cost_analytics", site, month, months, pc_daily: pcRows, prod_cost: costRows, totals: { total_cost: sumField(pcRows, "Total Cost"), feed_cost: sumField(pcRows, "Feed Cost"), labor_cost: sumField(pcRows, "Labor Cost"), utility_cost: sumField(pcRows, "Utility Cost") } }; } // 8. Quality & Energy (MCOS Daily) function getQualityEnergy(site, week) { const sheet = getSheet(SHEET.MCOS); let rows = sheetToObjects(sheet); const weeks = uniqueWeeks(rows); if (!week) week = weeks[weeks.length - 1] || ""; rows = filterByWeek(filterBySite(rows, site), week); return { tab: "quality_energy", site, week, weeks, rows, totals: { energy_kwh: sumField(rows, "Energy KWH") || sumField(rows, "KWH"), energy_cost: sumField(rows, "Energy Cost"), quality_score: rows.length ? (sumField(rows, "Quality Score") / rows.length).toFixed(2) : 0, rejection_rate: rows.length ? (sumField(rows, "Rejection Rate") / rows.length).toFixed(2) : 0, mcos: rows.length ? (sumField(rows, "MCOS") / rows.length).toFixed(4) : 0 } }; } // ─── METADATA ENDPOINT ─────────────────────────────────────────────────────── // Called with ?tab=meta to verify sheet connectivity function getMeta() { const results = {}; Object.entries(SHEET).forEach(([key, name]) => { try { const s = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name); results[key] = s ? { found: true, rows: s.getLastRow(), cols: s.getLastColumn() } : { found: false, error: "Sheet not found: " + name }; } catch (e) { results[key] = { found: false, error: e.message }; } }); return { tab: "meta", sheets: results, sites: SITES, version: "v39" }; }
